export function getSpotlightProfile(slug?: string): SpotlightProfile {

    return SPOTLIGHT_PROFILES[slug];

}

export function getOtherSpotlightProfiles(currentSlug: string): SpotlightProfile[] {

    .filter((p) => p.slug !== currentSlug)
    .slice(0, 3);
}
